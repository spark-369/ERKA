# --- Vector storage (embedded + persistent, no server required) --------------

# chromadb declares `posthog>=2.4.0` with NO upper bound, but its telemetry
# client calls the posthog v2/v3 positional API:
#     posthog.capture(distinct_id, event_name, properties)
# posthog >= 4 changed that to `capture(event, **kwargs)`, so an unbounded
# resolve installs an incompatible major and every Chroma event logs
# "capture() takes 1 positional argument but 3 were given".
#
# The application does not rely on this pin: `app/db/vector_store.py`
# replaces Chroma's telemetry client with a no-op via
# `chroma_product_telemetry_impl`, which removes the broken call regardless of
# the installed posthog version. The cap simply keeps the environment
# reproducible for anything else that imports chromadb.
posthog>=2.4,<4.0
# Imported directly by `app/db/vector_store.py` (Chroma's telemetry base class
# uses the `overrides` metaclass, which requires @override on the `capture` we
# replace). Already a chromadb dependency; declared explicitly because we now
# import it ourselves rather than relying on the transitive pin.
overrides>=7.3.1
